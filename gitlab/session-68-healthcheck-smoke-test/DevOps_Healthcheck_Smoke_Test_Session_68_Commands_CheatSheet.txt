# FROM python:3.12-slim

WORKDIR /app

RUN apt-get update \
    && apt-get install -y --no-install-recommends curl \
    && rm -rf /var/lib/apt/lists/*

COPY requirements.txt .

RUN python -m pip install \
    --no-cache-dir \
    -r requirements.txt

COPY . .

EXPOSE 5000

HEALTHCHECK \
    --interval=10s \
    --timeout=3s \
    --start-period=10s \
    --retries=3 \
    CMD curl -fsS http://localhost:5000/health || exit 1

CMD ["python", "app.py"]

Explanation:
Technical correction: the earlier Dockerfile healthcheck used curl without installing it. This corrected Dockerfile installs curl, installs Python dependencies, defines the healthcheck, and starts the Flask application.

Options:
- -y = --yes / --assume-yes (assume yes): Automatically answer yes to apt-get prompts.
- --no-install-recommends = do not install recommended packages: Install only required packages and explicit dependencies.
- && = AND operator: Run the next shell command only if the previous command succeeds.
- --no-cache-dir = no cache directory: Disable the pip download cache.
- -r = --requirement (requirement file): Install packages listed in requirements.txt.
- --interval=10s = healthcheck interval: Run the Docker healthcheck every 10 seconds.
- --timeout=3s = healthcheck timeout: Fail a healthcheck attempt if it runs longer than 3 seconds.
- --start-period=10s = startup grace period: Give the container 10 seconds before failed checks count toward unhealthy status.
- --retries=3 = retry count: Mark the container unhealthy after 3 consecutive failed checks.
- -f = --fail (fail): Make curl fail on HTTP response codes of 400 or higher.
- -s = --silent (silent): Hide curl's progress meter and error messages.
- -S = --show-error (show error): Show curl errors even in silent mode.
- || = OR operator: Run exit 1 only when the curl command fails.

# #!/usr/bin/env bash

set -e

URL="http://localhost:8088/health"
MAX_RETRIES=10
SLEEP_SECONDS=3

for i in $(seq 1 "$MAX_RETRIES"); do

  echo "Healthcheck attempt: $i/$MAX_RETRIES"

  if curl \
      --connect-timeout 3 \
      --max-time 5 \
      -fsS \
      "$URL"; then

    echo
    echo "Application is healthy"
    exit 0

  fi

  sleep "$SLEEP_SECONDS"

done

echo "Application healthcheck failed"
exit 1

Explanation:
Runs the reusable healthcheck script from the lesson. It retries the health URL, waits between attempts, and returns success only when the application responds successfully.

Options:
- #!/usr/bin/env bash = shebang: Run the script with Bash located through env.
- set -e = errexit option: Exit when an unhandled command returns a non-zero status.
- $(seq 1 "$MAX_RETRIES") = command substitution: Generate the loop attempt numbers.
- --connect-timeout 3 = connection timeout: Allow up to 3 seconds to establish the connection.
- --max-time 5 = maximum time: Allow up to 5 seconds for the entire curl operation.
- -f = --fail (fail): Make curl fail on HTTP response codes of 400 or higher.
- -s = --silent (silent): Hide curl's progress meter and error messages.
- -S = --show-error (show error): Show curl errors even in silent mode.
- exit 0 = success exit status: End the script successfully.

  stage: verify

  tags:
    - dev-shell

  script:
    - |
      for i in $(seq 1 10); do
        if ssh deploy@192.168.94.91 \
          "curl --connect-timeout 3 --max-time 5 -fsS http://localhost:8088/health"; then
