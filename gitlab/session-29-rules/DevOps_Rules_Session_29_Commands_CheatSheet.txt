# cat > .gitlab-ci.yml <<'EOF'
stages:
  - verify
  - test
  - build
  - deploy
  - release

branch-check:
  stage: verify
  tags:
    - dev-shell
  script:
    - echo "Pipeline Source = $CI_PIPELINE_SOURCE"
    - echo "Branch = $CI_COMMIT_BRANCH"
    - echo "Tag = $CI_COMMIT_TAG"
    - echo "Commit = $CI_COMMIT_SHORT_SHA"
  rules:
    - if: '$CI_PIPELINE_SOURCE == "push" && $CI_COMMIT_BRANCH'

mr-test:
  stage: test
  tags:
    - dev-shell
  script:
    - echo "Merge Request Pipeline detected"
    - echo "Source Branch = $CI_MERGE_REQUEST_SOURCE_BRANCH_NAME"
    - echo "Target Branch = $CI_MERGE_REQUEST_TARGET_BRANCH_NAME"
    - ./app/app.sh
  rules:
    - if: '$CI_PIPELINE_SOURCE == "merge_request_event"'

docs-check:
  stage: test
  tags:
    - dev-shell
  script:
    - echo "Documentation changed in Merge Request"
    - test -f docs/README.md
  rules:
    - if: '$CI_PIPELINE_SOURCE == "merge_request_event"'
      changes:
        - docs/**/*

docker-check:
  stage: test
  tags:
    - dev-shell
  script:
    - echo "Dockerfile exists"
    - test -f Dockerfile
  rules:
    - exists:
        - Dockerfile

build-main:
  stage: build
  tags:
    - dev-shell
  script:
    - echo "Building commit $CI_COMMIT_SHORT_SHA"
    - ./app/app.sh
  rules:
    - if: '$CI_COMMIT_BRANCH == $CI_DEFAULT_BRANCH'

deploy-production:
  stage: deploy
  tags:
    - dev-shell
  script:
    - echo "Production deployment simulation"
    - echo "Commit = $CI_COMMIT_SHORT_SHA"
  rules:
    - if: '$CI_COMMIT_BRANCH == $CI_DEFAULT_BRANCH'
      when: manual
      allow_failure: false

release:
  stage: release
  tags:
    - dev-shell
  script:
    - echo "Creating release for $CI_COMMIT_TAG"
  rules:
    - if: '$CI_COMMIT_TAG =~ /^v[0-9]+\.[0-9]+\.[0-9]+$/'
EOF

Explanation:
Creates .gitlab-ci.yml and writes the complete GitLab CI/CD rules lab configuration into it.

Options:
- > = output redirection: Create or overwrite .gitlab-ci.yml with the output from cat.
- <<'EOF' = quoted here-document: Pass the following lines to cat until EOF is reached, without shell expansion inside the here-document.
