# stages:
  - test
  - build
  - downstream

frontend_test:
  stage: test
  tags:
    - dev-shell
  script:
    - echo "Running frontend tests..."
    - echo "Frontend tests passed."

frontend_build:
  stage: build
  tags:
    - dev-shell
  script:
    - echo "Building frontend..."
    - echo "Frontend image tag = $CI_COMMIT_SHORT_SHA"

trigger_backend:
  stage: downstream

  variables:
    FRONTEND_PROJECT: "$CI_PROJECT_PATH"
    FRONTEND_COMMIT: "$CI_COMMIT_SHA"
    FRONTEND_IMAGE_TAG: "$CI_COMMIT_SHORT_SHA"

  trigger:
    project: cicd-lab/backend-app
    branch: main
    strategy: mirror

  rules:
    - if: '$CI_COMMIT_BRANCH == "main"'

Explanation:
This GitLab CI configuration tests and builds the frontend, then triggers the backend project on the main branch. `strategy: mirror` makes the trigger job reflect the downstream pipeline status.

# stages:
  - test
  - build
  - downstream

backend_test:
  stage: test
  tags:
    - dev-shell
  script:
    - echo "Running backend tests..."
    - echo "Frontend tag received = $FRONTEND_IMAGE_TAG"

backend_build:
  stage: build
  tags:
    - dev-shell
  script:
    - echo "Building backend..."
    - echo "Backend image tag = $CI_COMMIT_SHORT_SHA"

trigger_deployment:
  stage: downstream

  variables:
    FRONTEND_IMAGE_TAG: "$FRONTEND_IMAGE_TAG"
    BACKEND_IMAGE_TAG: "$CI_COMMIT_SHORT_SHA"
    BACKEND_PROJECT: "$CI_PROJECT_PATH"
    BACKEND_COMMIT: "$CI_COMMIT_SHA"

  trigger:
    project: cicd-lab/deployment-pipeline
    branch: main
    strategy: mirror

  rules:
    - if: '$CI_PIPELINE_SOURCE == "pipeline"'

Explanation:
This GitLab CI configuration tests and builds the backend, then triggers the deployment project. It forwards frontend and backend version information to the downstream deployment pipeline.
------------------------------------------------------------

# rules:
  - if: '$CI_PIPELINE_SOURCE == "pipeline"'

Explanation:
This GitLab CI rule allows the job only when the pipeline source is another pipeline. It is used to limit execution to multi-project downstream pipeline runs.

# stages:
  - validate
  - deploy

validate_versions:
  stage: validate
  tags:
    - dev-shell
  script:
    - echo "Frontend version = $FRONTEND_IMAGE_TAG"
    - echo "Backend version  = $BACKEND_IMAGE_TAG"

deploy_dev:
  stage: deploy
  tags:
    - dev-shell
  script:
    - echo "Deploying application..."
    - echo "Frontend = $FRONTEND_IMAGE_TAG"
    - echo "Backend  = $BACKEND_IMAGE_TAG"
    - echo "Target    = DEV-2"

  rules:
    - if: '$CI_PIPELINE_SOURCE == "pipeline"'

Explanation:
This deployment pipeline validates the received version variables and runs the deployment stage. The deploy job runs only when the pipeline was triggered by another pipeline.
