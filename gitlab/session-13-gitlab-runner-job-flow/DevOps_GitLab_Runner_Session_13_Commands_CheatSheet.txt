# cat > .gitlab-ci.yml <<'EOF'
stages:
  - inspect

runner_info:
  stage: inspect
  tags:
    - dev-shell
  script:
    - echo "===== RUNNER INFORMATION ====="
    - echo "CI_RUNNER_ID=$CI_RUNNER_ID"
    - echo "CI_RUNNER_DESCRIPTION=$CI_RUNNER_DESCRIPTION"
    - echo "CI_RUNNER_TAGS=$CI_RUNNER_TAGS"
    - echo "CI_RUNNER_VERSION=$CI_RUNNER_VERSION"
    - echo "CI_RUNNER_EXECUTABLE_ARCH=$CI_RUNNER_EXECUTABLE_ARCH"
    - echo "CI_PROJECT_DIR=$CI_PROJECT_DIR"
    - echo "===== OPERATING SYSTEM ====="
    - whoami
    - hostname
    - pwd
    - id
    - uname -a
EOF

Explanation:
Creates .gitlab-ci.yml and writes the complete GitLab CI configuration shown in the here-document into the file.

Options:
- > = shell output redirection: Redirects the output of cat into .gitlab-ci.yml and overwrites the file if it already exists.
- <<'EOF' = quoted here-document operator: Supplies all following lines to cat until the closing EOF line; quoting EOF prevents shell expansion inside the here-document.
