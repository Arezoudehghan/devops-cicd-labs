# cat > .gitlab-ci.yml <<'EOF'
stages:
  - validate
  - build
  - test
  - deploy

validate_app:
  stage: validate
  tags:
    - dev-shell
  script:
    - echo "Validating application..."
    - bash -n app.sh
    - echo "Validation completed successfully."

build_app:
  stage: build
  tags:
    - dev-shell
  script:
    - echo "Building application..."
    - mkdir -p build
    - cp app.sh build/app.sh
    - chmod +x build/app.sh
    - ./build/app.sh
    - echo "Build completed successfully."

syntax_test:
  stage: test
  tags:
    - dev-shell
  script:
    - echo "Running syntax test..."
    - bash -n app.sh
    - echo "Syntax test passed."

health_test:
  stage: test
  tags:
    - dev-shell
  script:
    - echo "Running application health test..."
    - ./app.sh | grep -q "healthy"
    - echo "Health test passed."

deploy_preview:
  stage: deploy
  tags:
    - dev-shell
  script:
    - echo "All previous stages passed."
    - echo "Deployment target would be DEV-2 at 192.168.94.91."
    - echo "Deployment preview completed."
EOF

Explanation:
Creates or overwrites .gitlab-ci.yml with the complete four-stage GitLab CI/CD pipeline used in this lab. The quoted here-document preserves the YAML content exactly while writing it.

Options:
- > = output redirection: Writes the output of cat to .gitlab-ci.yml, replacing the file if it already exists.
- <<'EOF' = quoted here-document: Supplies all lines up to EOF as standard input to cat without expanding variables or command substitutions.
